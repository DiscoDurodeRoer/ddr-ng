import {
  Component,
  inject,
  signal,
  WritableSignal
} from '@angular/core';
import { DdrButtonComponent } from 'ddr-ng/components/button';
import {
  DdrTranslatePipe,
  DdrTranslateService
} from 'ddr-ng/translate';

@Component({
  selector: 'translate-standalone',
  template: `<ddr-button 
    [text]="'change.language' | ddrTranslate"
    (action)="changeLanguage()"
/>
<p>{{'standalone.translation' | ddrTranslate}}</p>`,
  imports: [
    DdrButtonComponent,
    DdrTranslatePipe
  ],
  providers: [
    DdrTranslateService
  ]
})
export class ExampleComponent {

  private ddrTranslateService: DdrTranslateService = inject(DdrTranslateService);

  private language: WritableSignal<string> = signal<string>('en')

  constructor() {
    this.ddrTranslateService.getData('/data/i18n/translate-showcase/', 'en')
  }

  changeLanguage() {
    switch (this.language()) {
      case 'es':
        this.language.set('en');
        break;
      case 'en':
        this.language.set('es');
        break;
    }
    this.ddrTranslateService.getData('/data/i18n/translate-showcase/', this.language())
  }


}

import {
  Component,
  effect,
  inject
} from '@angular/core';
import { DdrButtonComponent } from 'ddr-ng/components/button';
import {
  DdrTranslateService,
  DdrTranslatePipe
} from 'ddr-ng/translate';

@Component({
  selector: 'app-example-component',
  template: `
    <div>
        <ddr-button text="Change to Spanish" (action)="changeToSpanish()" />
        <ddr-button text="Change to English" (action)="changeToEnglish()" />
    </div>
  `,
  imports: [
    DdrButtonComponent,
    DdrTranslatePipe
  ]
})
export class ExampleComponent {

  private ddrTranslateService: DdrTranslateService = inject(DdrTranslateService)

  constructor() {
    this.ddrTranslateService.getData('/data/i18n/translate-showcase/', 'en')
    effect(() => console.log(`Current language: ${this.ddrTranslateService.currentLanguage()}`))
  }

  changeToSpanish() {
    this.ddrTranslateService.changeLanguage('es')
  }

  changeToEnglish() {
    this.ddrTranslateService.changeLanguage('en')
  }
